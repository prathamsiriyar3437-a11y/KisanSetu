    // --- Dynamic Fare Estimation Engine (Slide 8) ---
    function calculateEstimate() {
      const cropSelect = document.getElementById('crop-select');
      const selectedOption = cropSelect.options[cropSelect.selectedIndex];
      const rateMultiplier = parseFloat(selectedOption.getAttribute('data-rate')) || 1.0;
      const isPerishable = selectedOption.getAttribute('data-perish') === 'high';

      const qty = parseFloat(document.getElementById('crop-qty').value) || 0;
      const reeferChecked = document.getElementById('require-reefer').checked;
      const aggregateChecked = document.getElementById('opt-aggregate').checked;

      // Auto-suggest reefer for high perishable
      if (isPerishable && !reeferChecked && cropSelect.value === 'Tomato') {
        document.getElementById('require-reefer').checked = true;
      }

      // Base Freight = Qty * Standard Route Base (₹0.85/kg) * Multiplier
      const baseFreight = qty * 0.93 * rateMultiplier;
      const handlingFee = document.getElementById('require-reefer').checked ? (qty * 0.35) : 0;
      
      // Shared aggregation rebate (approx 35-42% reduction)
      const discount = aggregateChecked ? (baseFreight * 0.38) : 0;
      const platformFee = Math.max(12, (baseFreight - discount) * 0.025);
      const total = Math.max(80, (baseFreight + handlingFee - discount + platformFee));

      document.getElementById('est-base-freight').innerText = `₹ ${baseFreight.toFixed(2)}`;
      document.getElementById('est-handling-fee').innerText = `₹ ${handlingFee.toFixed(2)}`;
      document.getElementById('est-discount').innerText = aggregateChecked ? `- ₹ ${discount.toFixed(2)}` : `₹ 0.00`;
      document.getElementById('est-platform-fee').innerText = `₹ ${platformFee.toFixed(2)}`;
      document.getElementById('est-total-cost').innerText = `₹ ${total.toFixed(2)}`;
      document.getElementById('est-per-kg').innerText = `₹ ${(total / qty).toFixed(2)} / kg`;

      appState.lastCalculatedCost = total;
    }

    // --- Farmer Form Submission Handler ---
    function handleFarmerSubmit(e) {
      e.preventDefault();
      const crop = document.getElementById('crop-select').value;
      const qty = document.getElementById('crop-qty').value;
      const pickup = document.getElementById('pickup-point').value;
      const dest = document.getElementById('dest-market').value;
      const mode = document.getElementById('access-mode-select').value;

      if (!appState.isOnline) {
        appState.offlineQueue.push({ crop, qty, pickup, dest, mode, timestamp: new Date() });
        document.getElementById('offline-queue-count').innerText = `${appState.offlineQueue.length} Queued`;
        showToast(`Offline mode: Booking saved locally! SMS backup sent to +91-XXXXXXXX.`);
        logAuditEvent(`[OFFLINE_ENQUEUE] Booking cached in IndexedDB. SMS payload dispatched via Twilio/CDAC gateway.`);
        return;
      }

      showToast(`Shipment for ${qty}kg ${crop} added to Smart Aggregation Pool!`);
      logAuditEvent(`[SHIPMENT_POSTED] ${qty}kg ${crop} from ${pickup} ➔ ${dest} [Mode: ${mode.toUpperCase()}]. Est Fare: ₹${appState.lastCalculatedCost.toFixed(2)}`);
      
      // Move user smoothly to Tab 2 to inspect aggregation
      setTimeout(() => {
        switchTab('aggregation-view');
      }, 700);
    }
